namespace = ate_cd

# The player writes the description of their culture (phase 8).
# Fired by on_culture_created for the player who created a hybrid/divergent
# culture, and by the "Edit Description" button of the culture window
# (culture head only, unlimited edits). Both pass the culture as
# scope:ate_cd_culture.
#
# The engine stores free text only on characters, so the text lives on the
# author and the culture points at them through var:ate_cd_text_holder.
# A character has ONE text key, so if root still authors another culture's
# text (e.g. the culture they diverged from), that text is handed over to
# that culture's head before root's key is reused.
# Engine rule (tested): the typed text is written to root with ANY option.
# "Keep" is therefore only a real cancel when root is not the culture's current
# author, so it is hidden when they are (editing your own text = Save only).
# See scripts/fase10_informe_texto_editable.md.
ate_cd.0001 = {
	type = character_event
	title = ate_cd.0001.t
	desc = ate_cd.0001.desc
	theme = default
	left_portrait = root

	trigger = {
		is_ai = no
	}

	immediate = {
		# Console or any caller that did not pass the culture
		if = {
			limit = { NOT = { exists = scope:ate_cd_culture } }
			culture = { save_scope_as = ate_cd_culture }
		}

		# 1. Free root's text key: another culture still showing root's text
		# keeps it through its current head, or goes back to its generic
		random_culture_global = {
			limit = {
				var:ate_cd_text_holder ?= root
				NOT = { this = scope:ate_cd_culture }
			}
			save_scope_as = ate_cd_previous_culture
		}
		if = {
			limit = { exists = scope:ate_cd_previous_culture }
			ate_cd_hand_over_description_effect = yes
		}

		# 2. Someone else wrote this culture's text (e.g. a dead predecessor):
		# copy it onto root so the field comes prefilled with it
		if = {
			limit = {
				scope:ate_cd_culture = {
					exists = var:ate_cd_text_holder
					NOT = { var:ate_cd_text_holder ?= root }
				}
			}
			scope:ate_cd_culture.var:ate_cd_text_holder = { save_scope_as = ate_cd_old_author }
			copy_localized_text = {
				key = ate_cd_custom_desc
				target = scope:ate_cd_old_author
			}
		}
	}

	widget = {
		gui = "ate_cd_enter_description"
		container = "custom_widgets_container"
		controller = {
			type = text
			data = {
				key = ate_cd_custom_desc
				default = ""
			}
		}
		setup_scope = {
			root = { save_scope_as = text_target }
		}
	}

	# Save: keep the text after the author's death and point the culture at them
	option = {
		name = ate_cd.0001.a
		store_localized_text_in_death = ate_cd_custom_desc
		# The previous author no longer needs to keep it (save size)
		if = {
			limit = { exists = scope:ate_cd_old_author }
			scope:ate_cd_old_author = { remove_localized_text = ate_cd_custom_desc }
		}
		scope:ate_cd_culture = {
			set_variable = {
				name = ate_cd_text_holder
				value = root
			}
		}
	}

	# Keep: the culture keeps pointing where it did. Hidden when root is already
	# the author, because their own text would be overwritten anyway.
	# ?= : the tooltip builder evaluates every line, so a plain = on an
	# unset var logs an error every frame while the event is open
	option = {
		name = ate_cd.0001.b
		trigger = {
			NOT = {
				scope:ate_cd_culture = { var:ate_cd_text_holder ?= root }
			}
		}
	}

	after = {
		# A picked default entry is stored as a variable with the widget key
		if = {
			limit = { has_variable = ate_cd_custom_desc }
			remove_variable = ate_cd_custom_desc
		}
	}
}
