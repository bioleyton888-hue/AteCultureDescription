# AteCultureDescription - scripted effects

# Gives a culture its description key.
# Never overwrites a real key (earlier registration, a hybrid/divergent generic
# assigned at creation, or a template the player picked). The only key it may
# replace is a base-culture generic (marked with ate_cd_key_is_base_generic),
# so a description written later always wins over the family generic.
# The GUI turns the flag into the loc key "<flag>_desc".
# $CULTURE$ = culture id, e.g. sunshiner -> loc key ate_cd_sunshiner_desc
ate_cd_register_culture_effect = {
	culture:$CULTURE$ = {
		if = {
			limit = {
				OR = {
					NOT = { has_variable = ate_cd_culture_key }
					has_variable = ate_cd_key_is_base_generic
				}
			}
			set_variable = { name = ate_cd_culture_key value = flag:ate_cd_$CULTURE$ }
			remove_variable = ate_cd_key_is_base_generic
		}
		# Empty check that only exists to "use" the flag: the GUI reads it, but CK3
		# ignores GUI/loc usage and logs "Flag ... is set but is never used"
		# (same workaround as Elder Kings 2's culture_key_setup).
		if = { limit = { var:ate_cd_culture_key = flag:ate_cd_$CULTURE$ } }
	}
}

# Gives a culture the generic key of its heritage family, by how it was born:
# hybrid, divergent, or base (an original ATE culture). Base generics are marked
# so a written description can replace them later (see above).
# Called by the generated ate_cd_assign_generic_key_effect (root = the culture).
# $FAMILY$ = family key from scripts/heritage_families.txt
ate_cd_set_generic_key_effect = {
	if = {
		limit = { is_hybrid_culture = yes }
		set_variable = { name = ate_cd_culture_key value = flag:ate_cd_generic_$FAMILY$_hybrid }
	}
	else_if = {
		limit = { is_divergent_culture = yes }
		set_variable = { name = ate_cd_culture_key value = flag:ate_cd_generic_$FAMILY$_divergent }
	}
	else = {
		set_variable = { name = ate_cd_culture_key value = flag:ate_cd_generic_$FAMILY$_base }
		set_variable = ate_cd_key_is_base_generic
	}
	# "use" the flags so CK3 does not log "Flag ... is set but is never used"
	# (see ate_cd_register_culture_effect)
	if = { limit = { var:ate_cd_culture_key = flag:ate_cd_generic_$FAMILY$_hybrid } }
	if = { limit = { var:ate_cd_culture_key = flag:ate_cd_generic_$FAMILY$_divergent } }
	if = { limit = { var:ate_cd_culture_key = flag:ate_cd_generic_$FAMILY$_base } }
}

# Everything the game start and the yearly pulse do: first the written
# descriptions, then a family generic for every culture still without a key
# (base cultures, and hybrids/divergents from saves older than the mod).
ate_cd_refresh_all_keys_effect = {
	ate_cd_register_all_cultures_effect = yes
	ate_cd_mark_written_descriptions_effect = yes
	every_culture_global = {
		limit = { NOT = { has_variable = ate_cd_culture_key } }
		ate_cd_assign_generic_key_effect = yes
	}
}

# Phase 8. Root = a player about to reuse their single text key (event
# ate_cd.0001). scope:ate_cd_previous_culture still shows root's text: hand it
# to that culture's current head, if their own key is free (they author no
# culture). Otherwise the culture goes back to its generic/community text.
ate_cd_hand_over_description_effect = {
	scope:ate_cd_previous_culture = {
		if = {
			limit = {
				exists = culture_head
				culture_head = {
					NOT = { this = root }
					NOT = {
						any_culture_global = { var:ate_cd_text_holder ?= prev }
					}
				}
			}
			culture_head = {
				copy_localized_text = {
					key = ate_cd_custom_desc
					target = root
				}
				store_localized_text_in_death = ate_cd_custom_desc
				save_scope_as = ate_cd_new_author
			}
			set_variable = {
				name = ate_cd_text_holder
				value = scope:ate_cd_new_author
			}
		}
		else = {
			remove_variable = ate_cd_text_holder
		}
	}
}
