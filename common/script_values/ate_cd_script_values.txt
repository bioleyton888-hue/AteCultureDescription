# AteCultureDescription - script values

# Read by the culture description component (root = the culture). The GUI has
# no direct "variable exists" check, so it asks for a mode:
#   2 = a player wrote this culture's description (the author still exists)
#   1 = the culture has a description key (written text or family generic)
#   0 = nothing yet (culture from another mod, or before the first pulse)
ate_cd_desc_mode_value = {
	value = 0
	if = {
		limit = { exists = var:ate_cd_text_holder }
		value = 2
	}
	else_if = {
		limit = { has_variable = ate_cd_culture_key }
		value = 1
	}
}

# Read by the "give us a better one" line (root = the culture): 1 = show it.
# Hidden under a player's own text and for cultures whose description someone
# already wrote (var set by ate_cd_mark_written_descriptions_effect).
ate_cd_show_feedback_hint_value = {
	value = 1
	if = {
		limit = {
			OR = {
				exists = var:ate_cd_text_holder
				has_variable = ate_cd_has_written_text
			}
		}
		value = 0
	}
}
